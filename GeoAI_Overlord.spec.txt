block_cipher = None

# تضمين الملفات اللازمة واستبعاد المفاتيح الخاصة
added_files = [
    ('public_key.pem', '.'),
    ('geoai_spatial.db', '.'),
    ('geoai_audit.db', '.'),
]

a = Analysis(
    ['main.py'],
    pathex=['.'],
    binaries=[],
    datas=added_files,
    hiddenimports=[
        'PyQt6',
        'PyQt6.QtCore',
        'PyQt6.QtGui',
        'PyQt6.QtWidgets',
        'cryptography',
        'numpy',
        'sqlite3',
    ],
    hookspath=[],

    win_no_prefer_redirects=False,
    win_private_assemblies=False,
    cipher=block_cipher,
    noarchive=False,
)

pyz = PYZ(a.pure, a.zipped_data, cipher=block_cipher)

exe = EXE(
    pyz,
    a.scripts,
    [],
    exclude_binaries=True,
    name='GeoAI_Overlord',
    debug=False,
    bootloader_ignore_signals=False,
